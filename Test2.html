const express = require('express'); const bodyParser = require('body-parser'); const cors = require('cors'); const app = express(); app.use(cors()); app.use(bodyParser.json()); let latestStats = { temp: "--", speed: "--", memory: "--", ip: "--", lastUpdated: "Waiting..." }; // Receiver Endpoint app.post('/update', (req, res) => { latestStats = { ...req.body, lastUpdated: new Date().toLocaleTimeString() }; console.log(`[${latestStats.lastUpdated}] Data Received`); res.sendStatus(200); }); // The Minimalist E-Paper Webview app.get('/display', (req, res) => { res.send(`

SYSTEM STATS

TEMP: ${latestStats.temp}

SPEED: ${latestStats.speed}

MEM: ${latestStats.memory}

IP: ${latestStats.ip}

Updated: ${latestStats.lastUpdated}

`); }); app.listen(3000, '0.0.0.0', () => { console.log("Bridge Online. Listening on Port 3000..."); });
